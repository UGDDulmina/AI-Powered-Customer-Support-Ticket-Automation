from __future__ import annotations

from datetime import datetime, timezone

from app.db.base import Base

from sqlalchemy.orm import Mapped, mapped_column, relationship
from sqlalchemy import Boolean, String, DateTime

from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from app.models.ticket import Ticket


class SupportAgent(Base):
    __tablename__ = "support_agents"

    id: Mapped[int] = mapped_column(primary_key=True)

    name: Mapped[str] = mapped_column(
        String(150),
        nullable=False
    )

    email: Mapped[str] = mapped_column(
        String(255),
        unique=True,
        nullable=False,
        index=True
    )
    is_active: Mapped[bool] = mapped_column(
        Boolean,
        default=True,
        nullable=False
    )
    created_at: Mapped[DateTime] = mapped_column(
        DateTime(timezone=True),
        default=datetime.now(timezone.utc),
        nullable=False
    )

    assigned_tickets = relationship(
        "Ticket",
        back_populates="assigned_agent"
    )

    password_hash: Mapped[str | None] = mapped_column(
        String(255),
        nullable=True
    )

    role: Mapped[str] = mapped_column(
        String(20),
        nullable=False,
        default="agent",
        server_default="agent"
    )

from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.orm import Session

from app.db.dependencies import get_db
from app.schemas.auth import AgentLoginRequest, TokenResponse
from app.services.auth_service import authenticate_agent
from app.core.security import create_access_token
from app.core.auth_dependencies import get_current_agent
from app.models.support_agent import SupportAgent
from app.schemas.auth import AuthenticatedAgentResponse

router = APIRouter(
    prefix="/auth",
    tags=["Authentication"]
)


@router.post(
    "/login",
    response_model=TokenResponse
)
def login(
    payload: AgentLoginRequest,
    db: Session = Depends(get_db)
):
    agent = authenticate_agent(
        db=db,
        email=payload.email,
        password=payload.password
    )

    if agent is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid email or password",
            headers={"WWW-Authenticate": "Bearer"}
        )

    access_token = create_access_token(
        agent_id=agent.id
    )

    return TokenResponse(
        access_token=access_token,
        token_type="bearer"
    )

@router.get("/me", response_model=AuthenticatedAgentResponse)
def get_my_profile(
    current_agent: SupportAgent = Depends(get_current_agent),
):
    return current_agent

from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy import select
from sqlalchemy.exc import IntegrityError
from sqlalchemy.orm import Session

from app.db.dependencies import get_db
from app.models.support_agent import SupportAgent
from app.schemas.agent_registration import (
    AgentRegistrationRequest,
    AgentRegistrationResponse,
)
from app.core.security import hash_password
from app.core.auth_dependencies import require_admin

router = APIRouter(
    prefix="/agents",
    tags=["Agent Registration"],
)


@router.post(
    "/register",
    response_model=AgentRegistrationResponse,
    status_code=status.HTTP_201_CREATED,
)
def register_agent(
    payload: AgentRegistrationRequest,
    db: Session = Depends(get_db),
    admin: SupportAgent = Depends(require_admin),
):
    email = str(payload.email).strip().lower()

    existing_agent = db.scalar(
        select(SupportAgent).where(SupportAgent.email == email)
    )

    if existing_agent:
        raise HTTPException(
            status_code=409,
            detail="Agent email already exists",
        )

    agent = SupportAgent(
        name=payload.name.strip(),
        email=email,
        password_hash=hash_password(payload.password),
        role=payload.role,
        is_active=True,
    )

    db.add(agent)

    try:
        db.commit()
        db.refresh(agent)
    except IntegrityError:
        db.rollback()
        raise HTTPException(
            status_code=409,
            detail="Agent email already exists",
        )

    return agent

@router.put(
    "/{agent_id}",
    response_model= SupportAgentResponse
)
def update_agent(
    agent_id: int,
    update_data: SupportAgentUpdate,
    database_session: Session = Depends(get_db)
):
    agent = database_session.get(SupportAgent, agent_id)

    if agent is None:
        raise HTTPException(
            status_code=404,
            detail="agent not found"
        )

    if update_data.email is not None:
        statement = select(SupportAgent).where(
           SupportAgent.email == update_data.email,
           SupportAgent.id != agent_id
        )

        dup_agent = database_session.scalar(statement)

        if dup_agent:
            raise HTTPException(
                status_code=409,
                detail="Duplicate email found."
            )

    updated_agent = update_data.model_dump(
        exclude_unset=True
    )

    for field_name,value in updated_agent.items():
        if value is None:
            raise HTTPException(
                status_code=422,
                detail=f"{field_name} cannot be null"
            )
        setattr(agent, field_name, value)

    try:
        database_session.commit()
        database_session.refresh(agent)

    except IntegrityError:
        database_session.rollback()
        raise HTTPException(
            status_code=409,
            detail="Agent update conflict with database constraints"
        )
    return agent

@router.get(
    "/workload",
    response_model=list[AgentWorkloadResponse]
)

def get_agent_workload(
    database_session: Session = Depends(get_db)
):
    statement = (
        select(
            SupportAgent,
            func.count(Ticket.id).label("active_tickets")
        ).outerjoin(
            Ticket,
            and_(
                Ticket.assigned_agent_id == SupportAgent.id,
                Ticket.status.in_(["open", "in_progress"])
            )
        ).group_by(SupportAgent.id).order_by(SupportAgent.id.asc())
    )

    results = database_session.execute(statement).all()

    workloads = []

    for row in results:
        agent = row.SupportAgent

        res = AgentWorkloadResponse(
            agent_id=agent.id,
            agent_name=agent.name,
            is_active=agent.is_active,
            active_tickets=row.active_tickets
        )

        workloads.append(res)

    return workloads

from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from sqlalchemy.orm import Session

from app.db.dependencies import get_db
from app.models.support_agent import SupportAgent
from app.core.security import decode_access_token

bearer_scheme = HTTPBearer()


def get_current_agent(
    credentials: HTTPAuthorizationCredentials = Depends(bearer_scheme),
    db: Session = Depends(get_db),
) -> SupportAgent:

    agent_id = decode_access_token(credentials.credentials)

    if agent_id is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid or expired access token",
        )

    agent = db.get(SupportAgent, agent_id)

    if agent is None or not agent.is_active:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Agent not found or inactive",
        )

    return agent


def require_admin(
    current_agent: SupportAgent = Depends(get_current_agent),
) -> SupportAgent:

    if current_agent.role != "admin":
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Admin access required",
        )

    return current_agent

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.models.support_agent import SupportAgent
from app.core.security import verify_password


def authenticate_agent(
    db: Session,
    email: str,
    password: str
) -> SupportAgent | None:

    normalized_email = email.strip().lower()

    agent = db.scalar(
        select(SupportAgent).where(
            SupportAgent.email == normalized_email
        )
    )

    if agent is None:
        return None

    if not agent.is_active:
        return None

    if not agent.password_hash:
        return None

    if not verify_password(password, agent.password_hash):
        return None

    return agent
