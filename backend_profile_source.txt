from fastapi import (
    APIRouter,
    Depends,
    HTTPException,
    status
)

from sqlalchemy import select, func
from sqlalchemy.exc import IntegrityError
from sqlalchemy.orm import Session

from app.db.dependencies import get_db

from app.schemas.auth import (
    AgentLoginRequest,
    TokenResponse,
    AuthenticatedAgentResponse,
    AgentProfileUpdateRequest,
    AgentChangePasswordRequest,
    PasswordChangeResponse
)

from app.services.auth_service import authenticate_agent

from app.core.security import (
    create_access_token,
    hash_password,
    verify_password
)

from app.core.auth_dependencies import get_current_agent
from app.models.support_agent import SupportAgent


router = APIRouter(
    prefix="/auth",
    tags=["Authentication"]
)


@router.post(
    "/login",
    response_model=TokenResponse
)
def login(
    payload: AgentLoginRequest,
    db: Session = Depends(get_db)
):
    agent = authenticate_agent(
        db=db,
        email=payload.email,
        password=payload.password
    )

    if agent is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid email or password",
            headers={"WWW-Authenticate": "Bearer"}
        )

    access_token = create_access_token(
        agent_id=agent.id
    )

    return TokenResponse(
        access_token=access_token,
        token_type="bearer"
    )


@router.get(
    "/me",
    response_model=AuthenticatedAgentResponse
)
def get_my_profile(
    current_agent: SupportAgent = Depends(
        get_current_agent
    )
):
    return current_agent


@router.patch(
    "/me",
    response_model=AuthenticatedAgentResponse
)
def update_my_profile(
    payload: AgentProfileUpdateRequest,
    db: Session = Depends(get_db),
    current_agent: SupportAgent = Depends(
        get_current_agent
    )
):
    normalized_email = payload.email.strip().lower()

    email_changed = (
        normalized_email != current_agent.email
    )

    if email_changed:
        if not payload.current_password:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="Current password is required to change email"
            )

        if (
            not current_agent.password_hash
            or not verify_password(
                payload.current_password,
                current_agent.password_hash
            )
        ):
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="Current password is incorrect"
            )

        existing_agent = db.scalar(
            select(SupportAgent).where(
                func.lower(SupportAgent.email)
                == normalized_email,
                SupportAgent.id != current_agent.id
            )
        )

        if existing_agent is not None:
            raise HTTPException(
                status_code=status.HTTP_409_CONFLICT,
                detail="Email address is already in use"
            )

    current_agent.name = payload.name
    current_agent.email = normalized_email

    try:
        db.commit()
        db.refresh(current_agent)

    except IntegrityError:
        db.rollback()

        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail="Unable to update profile. Email may already be in use"
        )

    return current_agent


@router.patch(
    "/change-password",
    response_model=PasswordChangeResponse
)
def change_my_password(
    payload: AgentChangePasswordRequest,
    db: Session = Depends(get_db),
    current_agent: SupportAgent = Depends(
        get_current_agent
    )
):
    if not current_agent.password_hash:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Account password is not configured"
        )

    if not verify_password(
        payload.current_password,
        current_agent.password_hash
    ):
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Current password is incorrect"
        )

    if verify_password(
        payload.new_password,
        current_agent.password_hash
    ):
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="New password must differ from current password"
        )

    current_agent.password_hash = hash_password(
        payload.new_password
    )

    db.commit()

    return PasswordChangeResponse(
        message="Password changed successfully"
    )

from pydantic import (
    BaseModel,
    EmailStr,
    Field,
    ConfigDict,
    field_validator
)


class AgentLoginRequest(BaseModel):
    email: EmailStr
    password: str


class TokenResponse(BaseModel):
    access_token: str
    token_type: str = "bearer"


class AuthenticatedAgentResponse(BaseModel):
    id: int
    name: str
    email: EmailStr
    role: str
    is_active: bool

    model_config = ConfigDict(
        from_attributes=True
    )


class AgentProfileUpdateRequest(BaseModel):
    name: str = Field(
        min_length=2,
        max_length=100
    )

    email: EmailStr

    current_password: str | None = Field(
        default=None,
        min_length=1
    )

    model_config = ConfigDict(
        extra="forbid"
    )

    @field_validator("name", mode="before")
    @classmethod
    def normalize_name(cls, value):
        if isinstance(value, str):
            return value.strip()

        return value

    @field_validator("email")
    @classmethod
    def normalize_email(cls, value):
        normalized = value.strip().lower()

        if len(normalized) > 255:
            raise ValueError(
                "Email cannot exceed 255 characters"
            )

        return normalized


class AgentChangePasswordRequest(BaseModel):
    current_password: str = Field(
        min_length=1
    )

    new_password: str = Field(
        min_length=12,
        max_length=128
    )

    model_config = ConfigDict(
        extra="forbid"
    )

    @field_validator("new_password")
    @classmethod
    def validate_new_password(cls, value):
        if not value.strip():
            raise ValueError(
                "Password cannot contain only spaces"
            )

        return value


class PasswordChangeResponse(BaseModel):
    message: str

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.models.support_agent import SupportAgent
from app.core.security import verify_password


def authenticate_agent(
    db: Session,
    email: str,
    password: str
) -> SupportAgent | None:

    normalized_email = email.strip().lower()

    agent = db.scalar(
        select(SupportAgent).where(
            SupportAgent.email == normalized_email
        )
    )

    if agent is None:
        return None

    if not agent.is_active:
        return None

    if not agent.password_hash:
        return None

    if not verify_password(password, agent.password_hash):
        return None

    return agent

from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from sqlalchemy.orm import Session

from app.db.dependencies import get_db
from app.models.support_agent import SupportAgent
from app.core.security import decode_access_token

bearer_scheme = HTTPBearer()


def get_current_agent(
    credentials: HTTPAuthorizationCredentials = Depends(bearer_scheme),
    db: Session = Depends(get_db),
) -> SupportAgent:

    agent_id = decode_access_token(credentials.credentials)

    if agent_id is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid or expired access token",
        )

    agent = db.get(SupportAgent, agent_id)

    if agent is None or not agent.is_active:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Agent not found or inactive",
        )

    return agent


def require_admin(
    current_agent: SupportAgent = Depends(get_current_agent),
) -> SupportAgent:

    if current_agent.role != "admin":
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Admin access required",
        )

    return current_agent

from __future__ import annotations

from datetime import datetime, timezone

from app.db.base import Base

from sqlalchemy.orm import Mapped, mapped_column, relationship
from sqlalchemy import Boolean, String, DateTime

from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from app.models.ticket import Ticket


class SupportAgent(Base):
    __tablename__ = "support_agents"

    id: Mapped[int] = mapped_column(primary_key=True)

    name: Mapped[str] = mapped_column(
        String(150),
        nullable=False
    )

    email: Mapped[str] = mapped_column(
        String(255),
        unique=True,
        nullable=False,
        index=True
    )
    is_active: Mapped[bool] = mapped_column(
        Boolean,
        default=True,
        nullable=False
    )
    created_at: Mapped[DateTime] = mapped_column(
        DateTime(timezone=True),
        default=datetime.now(timezone.utc),
        nullable=False
    )

    assigned_tickets = relationship(
        "Ticket",
        back_populates="assigned_agent"
    )

    password_hash: Mapped[str | None] = mapped_column(
        String(255),
        nullable=True
    )

    role: Mapped[str] = mapped_column(
        String(20),
        nullable=False,
        default="agent",
        server_default="agent"
    )
    @property
    def has_password(self) -> bool:
       return bool(self.password_hash)

from datetime import datetime, timedelta, timezone
import os

import jwt
from jwt.exceptions import InvalidTokenError
from pwdlib import PasswordHash
from dotenv import load_dotenv

load_dotenv()

password_hasher = PasswordHash.recommended()

JWT_SECRET_KEY = os.getenv("JWT_SECRET_KEY")
JWT_ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = int(
    os.getenv("ACCESS_TOKEN_EXPIRE_MINUTES", "30")
)


def hash_password(password: str) -> str:
    if not password:
        raise ValueError("Password cannot be empty")

    return password_hasher.hash(password)


def verify_password(
    plain_password: str,
    hashed_password: str
) -> bool:
    return password_hasher.verify(
        plain_password,
        hashed_password
    )


def create_access_token(
    agent_id: int,
    expires_delta: timedelta | None = None
) -> str:
    if not JWT_SECRET_KEY:
        raise RuntimeError("JWT_SECRET_KEY is not configured")

    now = datetime.now(timezone.utc)

    expire = now + (
        expires_delta
        if expires_delta is not None
        else timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    )

    payload = {
        "sub": str(agent_id),
        "iat": now,
        "exp": expire,
        "type": "access"
    }

    return jwt.encode(
        payload,
        JWT_SECRET_KEY,
        algorithm=JWT_ALGORITHM
    )


def decode_access_token(token: str) -> int | None:
    if not JWT_SECRET_KEY:
        raise RuntimeError("JWT_SECRET_KEY is not configured")

    try:
        payload = jwt.decode(
            token,
            JWT_SECRET_KEY,
            algorithms=[JWT_ALGORITHM],
            options={"require": ["sub", "exp", "iat"]}
        )

        if payload.get("type") != "access":
            return None

        agent_id = payload.get("sub")

        if not isinstance(agent_id, str):
            return None

        parsed_id = int(agent_id)

        if parsed_id <= 0:
            return None

        return parsed_id

    except (InvalidTokenError, ValueError, TypeError):
        return None
